  <script>
    const fallbackData = {
      "臺北市": { temp: 24, humidity: 78, wind: 3.2, weather: "多雲" },
      "臺中市": { temp: 27, humidity: 65, wind: 2.5, weather: "晴時多雲" },
      "高雄市": { temp: 30, humidity: 60, wind: 2.0, weather: "晴朗炎熱" }
    };

    function buildAdvice(data) {
      const tips = [];
      if (data.temp >= 30) {
        tips.push("氣溫達 30°C 以上，外出請注意防曬與補充水分。");
      } else if (data.temp <= 18) {
        tips.push("氣溫偏涼，建議攜帶薄外套或保暖衣物。");
      } else {
        tips.push("目前氣溫舒適，適合日常外出活動。");
      }

      if (data.humidity > 75) {
        tips.push("相對濕度大於 75%，體感較悶潮濕，留意除濕與通風。");
      }
      return tips.join(" ");
    }

    function renderWeather(city, data, modeText) {
      document.getElementById("tempVal").textContent = `${data.temp} °C`;
      document.getElementById("humVal").textContent = `${data.humidity} %`;
      document.getElementById("windVal").textContent = `${data.wind} m/s`;
      document.getElementById("adviceBox").textContent =
        `【${city}｜${data.weather}】${buildAdvice(data)}`;
      document.getElementById("statusText").textContent = `資料模式：${modeText}`;
    }

    async function fetchCwaObservation(apiKey, city) {
      // 示範呼叫中央氣象署觀測資料端點（O-A0003-001）；若環境阻擋或欄位不符，將自動回退至內建資料
      const url = `https://opendata.cwa.gov.tw/api/v1/rest/datastore/O-A0003-001?Authorization=${encodeURIComponent(apiKey)}`;
      const res = await fetch(url);
      if (!res.ok) throw new Error("API 回應狀態異常");
      const json = await res.json();
      const stations = json?.records?.Station || [];
      const matched = stations.find((st) => {
        const county = st?.GeoInfo?.CountyName || "";
        return county.includes(city.replace("台", "臺")) || county.includes(city.replace("臺", "台"));
      });
      if (!matched) throw new Error("查無對應縣市測站資料");

      const temp = Number(matched?.WeatherElement?.AirTemperature);
      const humidity = Number(matched?.WeatherElement?.RelativeHumidity);
      const wind = Number(matched?.WeatherElement?.WindSpeed);
      const weather = matched?.WeatherElement?.Weather || "即時觀測";

      if (Number.isNaN(temp) || Number.isNaN(humidity)) {
        throw new Error("氣象數值欄位解析失敗");
      }
      return {
        temp,
        humidity,
        wind: Number.isNaN(wind) ? 0 : wind,
        weather
      };
    }

    async function handleQuery() {
      const city = document.getElementById("citySelect").value;
      const apiKey = document.getElementById("apiKeyInput").value.trim();
      const fallback = fallbackData[city];

      if (!apiKey) {
        renderWeather(city, fallback, "內建示範資料");
        return;
      }

      document.getElementById("statusText").textContent = "資料模式：正在向氣象署 API 查詢...";
      try {
        const liveData = await fetchCwaObservation(apiKey, city);
        renderWeather(city, liveData, "中央氣象署即時觀測資料");
      } catch (err) {
        renderWeather(city, fallback, `改用內建示範資料（${err.message}）`);
      }
    }

    document.getElementById("queryBtn").addEventListener("click", handleQuery);
    window.addEventListener("DOMContentLoaded", handleQuery);
  </script>
